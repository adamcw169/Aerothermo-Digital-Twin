cmake_minimum_required(VERSION 3.16)
project(CrediTwinAero VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(creditwin_core cpp/src/thermal_model.cpp)
target_include_directories(creditwin_core PUBLIC cpp/include)
target_compile_options(creditwin_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(creditwin_sim cpp/src/main.cpp)
target_link_libraries(creditwin_sim PRIVATE creditwin_core)
add_executable(creditwin_cpp_tests cpp/tests/test_thermal.cpp)
target_link_libraries(creditwin_cpp_tests PRIVATE creditwin_core)
enable_testing()
add_test(NAME cpp_thermal_tests COMMAND creditwin_cpp_tests)
